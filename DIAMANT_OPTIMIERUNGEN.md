import * as Sharing from 'expo-sharing';
import * as FileSystem from 'expo-file-system';
import { captureRef } from 'react-native-view-shot';

export async function shareAchievement(
  achievement: Achievement,
  viewRef: React.RefObject<View>
) {
  try {
    // Capture view as image
    const uri = await captureRef(viewRef, {
      format: 'png',
      quality: 1,
    });

    // Share
    await Sharing.shareAsync(uri, {
      mimeType: 'image/png',
      dialogTitle: 'Teile dein Achievement!',
    });

    // Track share event
    await trackEvent('achievement_shared', { achievement: achievement.id });

    // Reward user
    await addXP(20); // Bonus XP für Sharing

    return true;
  } catch (error) {
    console.error('Share failed:', error);
    return false;
  }
}

export async function registerForPushNotifications() {
  if (!Device.isDevice) {
    alert('Push notifications only work on physical devices');
    return null;
  }

  const { status: existingStatus } = await Notifications.getPermissionsAsync();
  let finalStatus = existingStatus;

  if (existingStatus !== 'granted') {
    const { status } = await Notifications.requestPermissionsAsync();
    finalStatus = status;
  }

  if (finalStatus !== 'granted') {
    return null;
  }

  const token = (await Notifications.getExpoPushTokenAsync()).data;
  return token;
}

    await addPremiumDays(userId, days);
  } else if (rewardId.startsWith('voucher_')) {

  return reward;

export async function updateStreak(userId: number) {
  const user = await db.query.users.findFirst({
    where: eq(users.id, userId),
  });

  const today = new Date().toISOString().split('T')[0];
  const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0];

  if (user.lastActiveDate === today) {
    // Already logged in today
    return user.currentStreak;
  }

  let newStreak = 1;

  if (user.lastActiveDate === yesterday) {
    // Consecutive day
    newStreak = user.currentStreak + 1;
  }

  // Update streak
  await db.update(users).set({
    currentStreak: newStreak,
    longestStreak: Math.max(newStreak, user.longestStreak),
    lastActiveDate: today,
  }).where(eq(users.id, userId));

  // Record history
  await db.insert(streakHistory).values({
    userId,
    date: today,
    streakCount: newStreak,
  });

  // Milestone rewards
  if (newStreak === 7) {
    await addPoints(userId, POINT_VALUES.STREAK_7_DAYS, '7-Day Streak!');
  } else if (newStreak === 30) {
    await addPoints(userId, POINT_VALUES.STREAK_30_DAYS, '30-Day Streak!');
  } else if (newStreak === 100) {
    await addPoints(userId, POINT_VALUES.STREAK_100_DAYS, '100-Day Streak! 🔥');
  }

  return newStreak;
}
